cmake_minimum_required(VERSION 3.21)
project(MikuDanceStudioPortableEditor LANGUAGES CXX)
include(CTest)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(MDS_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(MIKUDANCESTUDIO_FORMATS_ONLY ON)
set(MIKUDANCESTUDIO_RHI_D3D9 OFF)
if(ANDROID)
    set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
option(MDS_EDITOR_QT "Build the Qt editor" ON)
option(MIKUDANCESTUDIO_RHI_VULKAN "Build Vulkan renderer" ON)
option(MIKUDANCESTUDIO_RHI_METAL "Build native macOS Metal renderer" ${APPLE})
add_subdirectory("${MDS_SOURCE}" production)
add_subdirectory("${MDS_SOURCE}/cmake/cpu-simulation-tests" cpu)
include("${MDS_SOURCE}/cmake/EditorCore.cmake")
mds_add_editor_core()
if(MDS_EDITOR_QT)
    include("${MDS_SOURCE}/cmake/QtEditor.cmake")
    mds_add_qt_editor()
endif()
